% Original figure content is never stretched or cropped. The locator is hidden.
\newcommand{\MaryFigure}[4]{%
  \begin{minipage}{\linewidth}\centering
    \if\relax\detokenize{#4}\relax
      \fbox{\parbox[c][.28\textheight][c]{.92\linewidth}{\centering #1}}%
    \else
      \includegraphics[width=\linewidth,height=.43\textheight,keepaspectratio]{#4}%
    \fi
    \par\smallskip{\footnotesize\textbf{#1}\par #3\par}%
  \end{minipage}%
}
